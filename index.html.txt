    <script>
        let currentPhoto = 1;
        let poppedCount = 0;

        function nextStep(stepNum) {
            document.querySelectorAll('.step-container').forEach(el => el.classList.remove('active'));
            document.getElementById('step' + stepNum).classList.add('active');
        }

        // STEP 1: RUNAWAY NO BUTTON
        function runAway() {
            const btn = document.getElementById('noBtn');
            const x = Math.random() * (window.innerWidth - 100) - (window.innerWidth / 2 - 50);
            const y = Math.random() * (window.innerHeight - 100) - (window.innerHeight / 2 - 50);
            btn.style.transform = `translate(${x}px, ${y}px)`;
        }

        // STEP 3: PHOTO UPLOADER
        function triggerUpload(num) {
            document.getElementById('file' + num).click();
        }

        function loadImg(event, num) {
            const img = document.getElementById('img' + num);
            const text = document.getElementById('ph' + num);
            img.src = URL.createObjectURL(event.target.files[0]);
            img.style.display = 'block';
            text.style.display = 'none';

            if (num === 1) setupPhoto(2, "to the chaos… 😂");
            else if (num === 2) setupPhoto(3, "to the moments we’ll always carry with us. ✨");
            else document.getElementById('toBalloonsBtn').style.display = 'inline-block';
        }

        function setupPhoto(num, caption) {
            document.getElementById('img1').parentElement.onclick = () => triggerUpload(num);
            document.getElementById('img1').style.display = 'none';
            document.getElementById('ph1').style.display = 'block';
            document.getElementById('cap1').innerText = caption;
            document.getElementById('file1').onchange = (e) => loadImg(e, num);
        }

        // STEP 4: POP BALLOONS
        function popBalloon(element, message) {
            if (element.style.visibility === 'hidden') return;
            element.style.visibility = 'hidden';
            
            const msgList = document.getElementById('balloonMsgs');
            const msgDiv = document.createElement('div');
            msgDiv.className = 'pop-msg';
            msgDiv.innerText = message;
            msgList.appendChild(msgDiv);

            poppedCount++;
            if (poppedCount === 4) {
                document.getElementById('toLetterBtn').style.display = 'inline-block';
            }
        }

        // STEP 6: BLOW CANDLE
        function blowCandle() {
            document.getElementById('flame').classList.add('off');
            document.getElementById('candleInstruction').innerText = "🥳 Your wish has been sent to the universe!";
            document.getElementById('toFinalBtn').style.display = 'inline-block';
        }
    </script>
